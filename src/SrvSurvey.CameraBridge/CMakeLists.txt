cmake_minimum_required(VERSION 3.24)
project(SrvSurveyCameraBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(CAMERA_BRIDGE_INTEGRATION_TESTS "Build a real Windows D3D11/DXGI/UDP harness" OFF)

if(BUILD_TESTING)
  add_executable(CameraProtocolTests tests/CameraProtocolTests.cpp)
  add_test(NAME CameraProtocol COMMAND CameraProtocolTests)
  add_executable(CaptureLayoutTests tests/CaptureLayoutTests.cpp)
  add_test(NAME CaptureLayout COMMAND CaptureLayoutTests)
  add_executable(CockpitReferenceTests tests/CockpitReferenceTests.cpp)
  add_test(NAME CockpitReference COMMAND CockpitReferenceTests)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The camera prototype requires Windows x64 (including Wine/DXVK)")
  endif()
  add_library(SrvSurveyCameraBridge SHARED CameraBridge.cpp CameraBridge.def)
  set_target_properties(SrvSurveyCameraBridge PROPERTIES PREFIX "")
  target_compile_definitions(SrvSurveyCameraBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(SrvSurveyCameraBridge PRIVATE ws2_32 dxguid uuid)
  if(CAMERA_BRIDGE_INTEGRATION_TESTS)
    add_executable(CameraProfileFilesTests tests/CaptureProfileFileTests.cpp)
    target_compile_definitions(CameraProfileFilesTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(BUILD_TESTING)
      add_test(NAME CameraProfileFiles COMMAND CameraProfileFilesTests "${CMAKE_CURRENT_SOURCE_DIR}/Profiles/EliteOdyssey-RigidCockpit.ini")
    endif()
    if(MSVC)
      target_compile_options(CameraProfileFilesTests PRIVATE /W4 /WX /EHsc)
      set_property(TARGET CameraProfileFilesTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
      target_compile_options(CameraProfileFilesTests PRIVATE -Wall -Wextra -Werror)
      target_link_options(CameraProfileFilesTests PRIVATE -static -municode)
    endif()
    add_executable(CameraBridgeWindowsIntegration tests/WindowsIntegration.cpp)
    target_compile_definitions(CameraBridgeWindowsIntegration PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(CameraBridgeWindowsIntegration PRIVATE ws2_32 dxguid uuid)
    if(MSVC)
      target_compile_options(CameraBridgeWindowsIntegration PRIVATE /W4 /WX /EHsc)
      set_property(TARGET CameraBridgeWindowsIntegration PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
      target_compile_options(CameraBridgeWindowsIntegration PRIVATE -Wall -Wextra -Werror)
      target_link_options(CameraBridgeWindowsIntegration PRIVATE -static -municode)
    endif()
  endif()
  if(MSVC)
    target_compile_options(SrvSurveyCameraBridge PRIVATE /W4 /WX /EHsc)
    set_property(TARGET SrvSurveyCameraBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(SrvSurveyCameraBridge PRIVATE -Wall -Wextra -Werror)
    target_link_options(SrvSurveyCameraBridge PRIVATE -static)
    if(BUILD_TESTING)
      target_link_options(CameraProtocolTests PRIVATE -static)
      target_link_options(CockpitReferenceTests PRIVATE -static)
      target_link_options(CaptureLayoutTests PRIVATE -static)
    endif()
  endif()
  install(TARGETS SrvSurveyCameraBridge RUNTIME DESTINATION .)
  install(FILES SrvSurveyCameraBridge.ini Install-CameraBridge.ps1 README.md DESTINATION .)
  install(DIRECTORY Profiles DESTINATION .)
endif()

if(MSVC AND BUILD_TESTING)
  foreach(test_target CameraProtocolTests CaptureLayoutTests CockpitReferenceTests)
    target_compile_options(${test_target} PRIVATE /W4 /WX /EHsc)
    set_property(TARGET ${test_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endforeach()
endif()

if(NOT MSVC AND BUILD_TESTING)
  target_compile_options(CaptureLayoutTests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(CameraProtocolTests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(CockpitReferenceTests PRIVATE -Wall -Wextra -Werror)
endif()
