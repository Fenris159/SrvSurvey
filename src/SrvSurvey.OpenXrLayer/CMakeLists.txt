cmake_minimum_required(VERSION 3.24)
project(SrvSurveyOpenXrLayer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(FrameProtocolTests tests/FrameProtocolTests.cpp)
  add_test(NAME FrameProtocol COMMAND FrameProtocolTests)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The SrvSurvey OpenXR layer requires Windows x64")
  endif()
  include(FetchContent)
  FetchContent_Declare(openxr_headers
    URL https://codeload.github.com/KhronosGroup/OpenXR-SDK/tar.gz/refs/tags/release-1.1.63
    URL_HASH SHA256=5b8dac0608f49498fe8e95f562fe6bcf0bc781e561f137348a8c251d06a88b6f
    SOURCE_SUBDIR srvsurvey-headers-only)
  FetchContent_MakeAvailable(openxr_headers)
  add_library(SrvSurveyOpenXrLayer SHARED Layer.cpp)
  target_include_directories(SrvSurveyOpenXrLayer PRIVATE ${openxr_headers_SOURCE_DIR}/include)
  target_compile_definitions(SrvSurveyOpenXrLayer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_NO_PROTOTYPES)
  target_link_libraries(SrvSurveyOpenXrLayer PRIVATE d3d11 dxgi shell32 ole32 uuid)
  if(BUILD_TESTING)
    add_executable(LayerTests tests/LayerTests.cpp)
    target_include_directories(LayerTests PRIVATE ${openxr_headers_SOURCE_DIR}/include)
    target_compile_definitions(LayerTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_NO_PROTOTYPES)
    target_link_libraries(LayerTests PRIVATE d3d11 dxgi shell32 ole32 uuid)
    add_test(NAME OpenXrLayer COMMAND LayerTests)
  endif()
  set_target_properties(SrvSurveyOpenXrLayer PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(SrvSurveyOpenXrLayer PRIVATE /W4 /WX /EHsc)
    set_property(TARGET SrvSurveyOpenXrLayer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(SrvSurveyOpenXrLayer PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    target_link_options(SrvSurveyOpenXrLayer PRIVATE -static)
    if(BUILD_TESTING)
      target_link_options(LayerTests PRIVATE -static)
      target_link_options(FrameProtocolTests PRIVATE -static)
    endif()
  endif()
  install(TARGETS SrvSurveyOpenXrLayer RUNTIME DESTINATION .)
  install(FILES SrvSurveyOpenXrLayer.json Install-OpenXrLayer.ps1 DESTINATION .)
  install(FILES ${openxr_headers_SOURCE_DIR}/LICENSE DESTINATION . RENAME Khronos-OpenXR-LICENSE)
endif()
