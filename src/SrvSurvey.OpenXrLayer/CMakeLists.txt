cmake_minimum_required(VERSION 3.24)
project(SrvSurveyOpenXrLayer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(FrameProtocolTests tests/FrameProtocolTests.cpp)
  add_test(NAME FrameProtocol COMMAND FrameProtocolTests)
  add_executable(PointerInteractionTests tests/PointerInteractionTests.cpp)
  add_test(NAME PointerInteraction COMMAND PointerInteractionTests)
endif()
if(WIN32 OR (UNIX AND NOT APPLE))
  include(FetchContent)
  FetchContent_Declare(openxr_headers
    URL https://codeload.github.com/KhronosGroup/OpenXR-SDK/tar.gz/refs/tags/release-1.1.63
    URL_HASH SHA256=5b8dac0608f49498fe8e95f562fe6bcf0bc781e561f137348a8c251d06a88b6f
    SOURCE_SUBDIR srvsurvey-headers-only)
  FetchContent_MakeAvailable(openxr_headers)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The SrvSurvey OpenXR layer requires Windows x64")
  endif()
  add_library(SrvSurveyOpenXrLayer SHARED Layer.cpp)
  target_include_directories(SrvSurveyOpenXrLayer PRIVATE ${openxr_headers_SOURCE_DIR}/include)
  target_compile_definitions(SrvSurveyOpenXrLayer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_NO_PROTOTYPES)
  target_link_libraries(SrvSurveyOpenXrLayer PRIVATE d3d11 dxgi shell32 ole32 uuid)
  if(BUILD_TESTING)
    add_executable(LayerTests tests/LayerTests.cpp)
    target_include_directories(LayerTests PRIVATE ${openxr_headers_SOURCE_DIR}/include)
    target_compile_definitions(LayerTests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_NO_PROTOTYPES)
    target_link_libraries(LayerTests PRIVATE d3d11 dxgi shell32 ole32 uuid)
    add_test(NAME OpenXrLayer COMMAND LayerTests)
  endif()
  set_target_properties(SrvSurveyOpenXrLayer PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(SrvSurveyOpenXrLayer PRIVATE /W4 /WX /EHsc)
    set_property(TARGET SrvSurveyOpenXrLayer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(SrvSurveyOpenXrLayer PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    target_link_options(SrvSurveyOpenXrLayer PRIVATE -static)
    if(BUILD_TESTING)
      target_link_options(LayerTests PRIVATE -static)
      target_link_options(FrameProtocolTests PRIVATE -static)
      target_link_options(PointerInteractionTests PRIVATE -static)
    endif()
  endif()
  install(TARGETS SrvSurveyOpenXrLayer RUNTIME DESTINATION .)
  install(FILES SrvSurveyOpenXrLayer.json Install-OpenXrLayer.ps1 DESTINATION .)
  install(FILES ${openxr_headers_SOURCE_DIR}/LICENSE DESTINATION . RENAME Khronos-OpenXR-LICENSE)
elseif(UNIX AND NOT APPLE)
  find_package(Threads REQUIRED)
  find_path(SRVSURVEY_VULKAN_INCLUDE_DIR vulkan/vulkan.h)
  if(NOT SRVSURVEY_VULKAN_INCLUDE_DIR)
    FetchContent_Declare(vulkan_headers
      URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/refs/tags/v1.3.296
      URL_HASH SHA256=e204e0b3c19f622d197df945737f5db913d6621830999b8578d34e80a7c90585
      SOURCE_SUBDIR srvsurvey-headers-only)
    FetchContent_MakeAvailable(vulkan_headers)
    set(SRVSURVEY_VULKAN_INCLUDE_DIR "${vulkan_headers_SOURCE_DIR}/include")
  endif()
  add_library(SrvSurveyLinuxOpenXrOverlay SHARED LinuxOverlay.cpp)
  target_include_directories(SrvSurveyLinuxOpenXrOverlay PRIVATE ${openxr_headers_SOURCE_DIR}/include ${SRVSURVEY_VULKAN_INCLUDE_DIR})
  target_compile_definitions(SrvSurveyLinuxOpenXrOverlay PRIVATE VK_NO_PROTOTYPES XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_VULKAN)
  target_link_libraries(SrvSurveyLinuxOpenXrOverlay PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
  target_compile_options(SrvSurveyLinuxOpenXrOverlay PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  if(BUILD_TESTING)
    add_executable(LinuxOverlayTests tests/LinuxOverlayTests.cpp LinuxOverlay.cpp)
    target_include_directories(LinuxOverlayTests PRIVATE ${openxr_headers_SOURCE_DIR}/include ${SRVSURVEY_VULKAN_INCLUDE_DIR})
    target_compile_definitions(LinuxOverlayTests PRIVATE VK_NO_PROTOTYPES XR_NO_PROTOTYPES XR_USE_GRAPHICS_API_VULKAN)
    target_link_libraries(LinuxOverlayTests PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    target_compile_options(LinuxOverlayTests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    add_test(NAME LinuxOpenXrOverlay COMMAND LinuxOverlayTests)
  endif()
  install(TARGETS SrvSurveyLinuxOpenXrOverlay LIBRARY DESTINATION .)
  install(FILES ${openxr_headers_SOURCE_DIR}/LICENSE DESTINATION . RENAME Khronos-OpenXR-LICENSE)
endif()
